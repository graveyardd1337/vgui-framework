cmake_minimum_required(VERSION 3.20)
project(vgui-framework LANGUAGES CXX)
option(VGUI_STATIC_RUNTIME "Link the MSVC runtime statically for standalone demos" OFF)
option(VGUI_BUILD_SHARED "Build vgui-framework as a DLL" OFF)
if(VGUI_BUILD_SHARED AND VGUI_STATIC_RUNTIME)
  message(FATAL_ERROR "DLL builds use the shared runtime; omit VGUI_STATIC_RUNTIME")
endif()
if(MSVC AND VGUI_STATIC_RUNTIME)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(MSVC AND DEFINED VGUI_MSVC_INCLUDE_PREFIX_FILE)
  # Some hosts decode localized /showIncludes output with the wrong code page.
  # build.ps1 detects the real prefix so Ninja tracks header dependencies correctly.
  file(READ "${VGUI_MSVC_INCLUDE_PREFIX_FILE}" CMAKE_CL_SHOWINCLUDES_PREFIX)
endif()
if(NOT WIN32)
  message(FATAL_ERROR "vgui-framework requires Windows and DirectX 11")
endif()
set(VGUI_LIBRARY_TYPE STATIC)
if(VGUI_BUILD_SHARED)
  set(VGUI_LIBRARY_TYPE SHARED)
endif()
add_library(vgui-framework ${VGUI_LIBRARY_TYPE}
  src/context.cpp
  src/input.cpp
  src/layout.cpp
  src/render.cpp
  src/widgets.cpp
)
add_library(vgui::vgui ALIAS vgui-framework)
if(VGUI_BUILD_SHARED)
  target_compile_definitions(vgui-framework PUBLIC VGUI_SHARED PRIVATE VGUI_BUILDING_LIBRARY)
endif()
target_include_directories(vgui-framework PUBLIC include)
target_compile_features(vgui-framework PUBLIC cxx_std_17)
target_compile_definitions(vgui-framework PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(vgui-framework PUBLIC d3d11 d3dcompiler dxgi gdi32 user32)
if(MSVC)
  target_compile_options(vgui-framework PRIVATE /W4 /permissive-)
endif()
add_executable(vgui_demo WIN32 demo/main.cpp)
target_link_libraries(vgui_demo PRIVATE vgui-framework)
include(CTest)
if(BUILD_TESTING)
  add_executable(vgui_interaction_tests tests/interaction.cpp)
  target_link_libraries(vgui_interaction_tests PRIVATE vgui-framework)
  add_test(NAME widget_interactions COMMAND vgui_interaction_tests)
  add_executable(vgui_api_tests tests/api.cpp examples/settings.cpp)
  target_link_libraries(vgui_api_tests PRIVATE vgui::vgui)
  add_test(NAME public_api COMMAND vgui_api_tests)
  add_test(NAME demo_smoke COMMAND vgui_demo --smoke)
  add_test(NAME demo_framed_smoke COMMAND vgui_demo --smoke --framed)
endif()
